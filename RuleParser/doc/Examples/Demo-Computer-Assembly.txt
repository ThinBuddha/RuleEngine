Enforce model "Demo-Computer-Assembly-Types.txt"

RuleContainer Computer.Order

  Rule initialize.build
  Priority 1
  If Build.Profile <> ""
  Then Build.Status := "collecting components"
     and CPU.Status := "pending"
     and Memory.Status := "pending"
     and Graphics.Status := "pending"
     and Storage.Status := "pending"
     and Power.Status := "pending"
     and Chassis.Status := "pending"
     and Cooling.Status := "pending"
     and Platform.Status := "pending"
     and System.Status := "pending"
     and Tests.Status := "pending"
     and Software.Status := "pending"
     and Build.Component.list := ""
     and Build.Component.count := 0
     and Build.Total.credits := 0


RuleContainer Computer.Assembly

  Rule release.computer
  Priority 1
  If Software.Status = "installed" and Tests.Status = "passed"
  Then Build.Status := "ready for delivery"

  Rule install.software
  Priority 2
  If Tests.Status = "passed"
  Then Software.Status := "installed"
     and Software.Package := "Demo Desktop OS"
     and Build.Total.credits := Build.Total.credits + 80

  Rule test.computer
  Priority 3
  If System.Status = "assembled" and Cooling.Status = "installed"
  Then Tests.Status := "passed"

  Rule assemble.system
  Priority 4
  If Platform.Status = "ready" and Graphics.Status = "installed" and Storage.Status = "installed" and Power.Status = "installed"
  Then System.Status := "assembled"

  Rule assemble.platform
  Priority 5
  If CPU.Status = "installed" and Memory.Status = "installed" and Chassis.Status = "installed"
  Then Platform.Status := "ready"
