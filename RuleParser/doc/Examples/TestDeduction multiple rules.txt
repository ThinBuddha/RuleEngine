RuleContainer Determine.how.to.travel

Rule walking.1
If not Outside.it.rains and
   Distance.to.travel = "small"
then 
   Travel.by.walking

Rule walking.2
If not Outside.it.rains and
   Distance.to.travel = "medium" and
   Stamina.of.person = "great"
then 
   Travel.by.walking

Rule walking.3
If Outside.it.rains and
   Distance.to.travel = "small" and
   Person.has.umbrella
then 
   Travel.by.walking

Rule walking.4
If Outside.it.rains and
   Distance.to.travel = "medium" and
   Person.has.umbrella
then 
   Travel.by.walking

Rule car.1
If Outside.it.rains and
   Distance.to.travel = "big" and
   Person.has.car and
   Price.of.gas = "good"
then 
   Travel.by.car

Rule train.1
If Outside.it.rains and
   Distance.to.travel = "big" and
   Person.has.car and
   Price.of.gas = "bad"
then 
   Travel.by.train

Rule train.2
If Outside.it.rains and
   Distance.to.travel = "big" and
   not Person.has.car 
then 
   Travel.by.train

Rule price.of.gas.1
If Travel.country = "Netherlands" 
then 
   Price.of.gas := "bad"

Rule price.of.gas.2
If Travel.country <> "Netherlands" 
then 
   Price.of.gas := "good"
